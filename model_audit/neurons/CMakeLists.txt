cmake_minimum_required(VERSION 3.20)
project(NeuronAudit LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
get_filename_component(STUDIO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(neuron_probe probe.cpp)
target_include_directories(neuron_probe PRIVATE "${STUDIO}")
target_link_libraries(neuron_probe PRIVATE "${STUDIO}/build/native/network_core.lib")
if(MSVC)
  target_compile_options(neuron_probe PRIVATE /utf-8 /W3 /O2)
endif()
