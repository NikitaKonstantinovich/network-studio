cmake_minimum_required(VERSION 3.20)
project(AstrocyteModelAudit LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
get_filename_component(STUDIO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(astrocyte_audit harness.cpp)
target_include_directories(astrocyte_audit PRIVATE "${STUDIO}")
target_link_libraries(astrocyte_audit PRIVATE "${STUDIO}/build/native/network_core.lib")
if(MSVC)
  target_compile_options(astrocyte_audit PRIVATE /utf-8 /W3)
endif()
