cmake_minimum_required(VERSION 3.20)
project(Lab3FtzDiagnostic LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
get_filename_component(STUDIO "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
add_executable(network_studio_cli_ftz launcher.cpp)
target_include_directories(network_studio_cli_ftz PRIVATE "${STUDIO}")
target_link_libraries(network_studio_cli_ftz PRIVATE "${STUDIO}/build/native/network_core.lib")
if(MSVC)
  target_compile_options(network_studio_cli_ftz PRIVATE /utf-8 /W3)
endif()
